cmake_minimum_required(VERSION 3.8)
project(cleany_axis_controller)
find_package(ament_cmake REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(dwb_msgs REQUIRED)
find_package(dwb_core REQUIRED)
find_package(dwb_plugins REQUIRED)
find_package(nav2_util REQUIRED)
find_package(nav_2d_msgs REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav2_behavior_tree REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(rclcpp_action REQUIRED)
add_library(axis_generator SHARED src/axis_generator.cpp)
target_compile_features(axis_generator PUBLIC cxx_std_17)
target_include_directories(axis_generator PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(axis_generator geometry_msgs dwb_msgs dwb_core dwb_plugins nav2_util nav_2d_msgs pluginlib rclcpp)
add_library(cleany_yield_wait_bt_node SHARED src/yield_wait.cpp)
target_include_directories(cleany_yield_wait_bt_node PRIVATE include)
target_compile_features(cleany_yield_wait_bt_node PUBLIC cxx_std_17)
target_compile_definitions(cleany_yield_wait_bt_node PRIVATE BT_PLUGIN_EXPORT)
ament_target_dependencies(cleany_yield_wait_bt_node nav2_behavior_tree nav2_msgs rclcpp)
install(TARGETS cleany_yield_wait_bt_node LIBRARY DESTINATION lib)
pluginlib_export_plugin_description_file(dwb_core plugins.xml)
install(TARGETS axis_generator EXPORT axis_generatorTargets LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_axis_generator test/test_axis_generator.cpp)
  target_link_libraries(test_axis_generator axis_generator)
  ament_target_dependencies(test_axis_generator dwb_plugins nav2_util rclcpp)
  ament_add_gtest(test_yield_wait test/test_yield_wait.cpp)
  target_include_directories(test_yield_wait PRIVATE include)
  ament_target_dependencies(test_yield_wait nav2_behavior_tree nav2_msgs rclcpp rclcpp_action)
endif()
ament_export_targets(axis_generatorTargets HAS_LIBRARY_TARGET)
ament_export_dependencies(geometry_msgs dwb_msgs dwb_core dwb_plugins nav2_util nav_2d_msgs pluginlib rclcpp)
ament_package()
