cmake_minimum_required(VERSION 3.16)
project(cleany_manipulation_bt)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(behaviortree_cpp 4 REQUIRED)
find_package(pybind11_vendor REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter Development)
find_package(pybind11 REQUIRED)
ament_python_install_package(${PROJECT_NAME})
pybind11_add_module(_bt_runner src/runner.cpp)
target_compile_features(_bt_runner PRIVATE cxx_std_17)
target_link_libraries(_bt_runner PRIVATE behaviortree_cpp::behaviortree_cpp)
install(TARGETS _bt_runner LIBRARY DESTINATION "${PYTHON_INSTALL_DIR}/${PROJECT_NAME}")
install(DIRECTORY trees config launch DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/manipulation_mujoco_server scripts/manipulation_jetson_server scripts/manipulation_recovery scripts/arm_return DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  # 파일별 실행으로 느린 검사와 종료 교착을 식별하며 전체 검사를 유지합니다.
  file(GLOB_RECURSE bt_test_files CONFIGURE_DEPENDS "test/test_*.py")
  foreach(test_file IN LISTS bt_test_files)
    get_filename_component(test_name "${test_file}" NAME_WE)
    ament_add_pytest_test(bt_${test_name} "${test_file}" TIMEOUT 60
      ENV "PYTEST_DISABLE_PLUGIN_AUTOLOAD=1")
  endforeach()
endif()
ament_package()
