cmake_minimum_required(VERSION 3.16)
project(cleany_manipulation_bt)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(behaviortree_cpp 4 REQUIRED)
find_package(pybind11_vendor REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter Development)
find_package(pybind11 REQUIRED)
ament_python_install_package(${PROJECT_NAME})
pybind11_add_module(_bt_runner src/runner.cpp)
target_compile_features(_bt_runner PRIVATE cxx_std_17)
target_link_libraries(_bt_runner PRIVATE behaviortree_cpp::behaviortree_cpp)
install(TARGETS _bt_runner LIBRARY DESTINATION "${PYTHON_INSTALL_DIR}/${PROJECT_NAME}")
install(DIRECTORY trees config launch DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/manipulation_mujoco_server DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(bt_backend test ENV "PYTEST_DISABLE_PLUGIN_AUTOLOAD=1")
endif()
ament_package()
