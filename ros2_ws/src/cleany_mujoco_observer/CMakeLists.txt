cmake_minimum_required(VERSION 3.16)
project(cleany_mujoco_observer)

find_package(ament_cmake REQUIRED)
find_package(mujoco_ros2_control_plugins REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(mujoco_vendor REQUIRED)
find_package(mujoco_ros2_control REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(glfw3 REQUIRED)

add_library(contact_observation STATIC src/contact_observation.cpp)
set_target_properties(contact_observation PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(contact_observation PUBLIC cxx_std_17)
target_include_directories(contact_observation PUBLIC include)
target_link_libraries(contact_observation PUBLIC mujoco::mujoco)

add_library(sorting_observer STATIC src/sorting_observer.cpp)
set_target_properties(sorting_observer PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(sorting_observer PUBLIC cxx_std_17)
ament_target_dependencies(sorting_observer
  mujoco_ros2_control_plugins pluginlib rclcpp visualization_msgs diagnostic_msgs)
target_link_libraries(sorting_observer contact_observation)
add_library(observed_mujoco_system SHARED src/observed_mujoco_system.cpp src/scheduled_cameras.cpp)
target_compile_features(observed_mujoco_system PUBLIC cxx_std_17)
target_compile_definitions(observed_mujoco_system PRIVATE ROS_DISTRO_HUMBLE=1)
ament_target_dependencies(observed_mujoco_system mujoco_ros2_control hardware_interface sensor_msgs rcl_interfaces)
target_link_libraries(observed_mujoco_system sorting_observer glfw)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_camera_rates test/test_camera_rates.cpp)
  target_include_directories(test_camera_rates PRIVATE include)
  ament_add_gtest(test_contact_observation test/test_contact_observation.cpp)
  target_link_libraries(test_contact_observation contact_observation)
  ament_add_gtest(test_geom_bounds test/test_geom_bounds.cpp)
  target_link_libraries(test_geom_bounds contact_observation)
  ament_add_gtest(test_observer_plugin test/test_observer_plugin.cpp)
  target_link_libraries(test_observer_plugin sorting_observer)
  ament_target_dependencies(test_observer_plugin
    mujoco_ros2_control_plugins pluginlib rclcpp diagnostic_msgs hardware_interface)
endif()
pluginlib_export_plugin_description_file(hardware_interface plugins.xml)
install(TARGETS observed_mujoco_system LIBRARY DESTINATION lib)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_package()
